# each test file is compiled on its own, so ninja builds them in parallel (one file holding
# every test took minutes to optimize). The small .cpp files are generated in the build folder,
# and are rewritten only when their text changes.
set(TEST_UNITS)
foreach(tdd IN LISTS TEST_FILES)
    string(REGEX REPLACE "[/.]" "_" unit "${tdd}")
    set(unit_cpp "${CMAKE_BINARY_DIR}/test_units/${unit}.cpp")
    file(CONFIGURE OUTPUT "${unit_cpp}"
         CONTENT "// generated by CMakeLists.txt: compiles ${tdd} on its own\n#include \"test/test_common.hpp\"\n#include \"${tdd}\"\n")
    list(APPEND TEST_UNITS "${unit_cpp}")
    set_source_files_properties("${tdd}" PROPERTIES HEADER_FILE_ONLY ON)
endforeach()

set(SOURCE_FILES
        source/catch_amalgamated.hpp
        source/catch_amalgamated.cpp
        ${TEST_FILES}
        ${TEST_UNITS}
        test/test_common.hpp
        source/jc313_Date.cpp
        source/jc313_Date.h
        source/jc_date_utils.hpp
        test_main.cpp
        )
add_executable(Date ${SOURCE_FILES})
target_include_directories(Date PRIVATE "${CMAKE_SOURCE_DIR}")
# -Wformat=2 -Werror=format: a printf-style call whose arguments do not match its format
#   (e.g. a 64-bit value given to %d) fails the build instead of misreading memory at run time
target_compile_options(Date PRIVATE -Wall -Wextra -Wformat=2 -Werror=format)
# Catch2 v3.8.1 (amalgamated); main() is in test_main.cpp, and Catch2's own code is not ours to warn about
target_compile_definitions(Date PRIVATE CATCH_AMALGAMATED_CUSTOM_MAIN)
set_source_files_properties(source/catch_amalgamated.cpp PROPERTIES COMPILE_OPTIONS "-w")

# main.cpp: a small day-of-week printout for 2017-2020 (not a test)
